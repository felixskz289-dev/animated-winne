function speak(text,rate=0.85){
  try{
    speechSynthesis.cancel();
    const u=new SpeechSynthesisUtterance(text);
    u.lang="zh-CN"; u.rate=rate;
    const v=speechSynthesis.getVoices().find(v=>v.lang.startsWith("zh"));
    if(v)u.voice=v;
    speechSynthesis.speak(u);
  }catch(e){}
}

  function shuffle(a){a=a.slice();for(let i=a.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[a[i],a[j]]=[a[j],a[i]];}return a;}
  function chars(zh){ // split, keeping punctuation attached
    return zh.match(/[，。！？、]|[\u4e00-\u9fff]+|\d+/g)?.join("").split(/(?=[\u4e00-\u9fff])|(?<=[\u4e00-\u9fff])|(?=[，。！？])/).filter(Boolean) || [];
  }
  function load(){
    locked=false; msg.textContent=""; msg.className="msg";
    const L=ORDER_LINES[idx];
    document.getElementById("woNum").textContent=idx+1;
    const parts=chars(L.zh);
    slots.innerHTML=""; tiles.innerHTML="";
    shuffle(parts.map((p,i)=>({p,id:i}))).forEach(t=>{
      const b=document.createElement("button"); b.className="tile"; b.dataset.p=t.p;
      b.innerHTML=`<span class="han">${t.p}</span>`;
      b.onclick=()=>{ if(locked)return;
        if(b.parentElement===tiles){ slots.appendChild(b); b.classList.add("slot-tile"); }
        else { tiles.appendChild(b); b.classList.remove("slot-tile"); }
      };
      tiles.appendChild(b);
    });
  }
  document.getElementById("woCheck").onclick=()=>{
    if(locked)return;
    const got=[...slots.children].map(b=>b.dataset.p).join("");
    const want=ORDER_LINES[idx].zh;
    if(got===want){
      locked=true; score++; document.getElementById("woScore").textContent=score;
      msg.textContent="✓ Отлично! "+ORDER_LINES[idx].py+" — "+ORDER_LINES[idx].ru;
      msg.className="msg ok"; speak(want);
      setTimeout(()=>{ idx=(idx+1)%ORDER_LINES.length; load(); },2200);
    } else {
      msg.textContent="✗ Пока не так. Подсказка: "+ORDER_LINES[idx].py;
      msg.className="msg bad";
    }
  };
  document.getElementById("woSkip").onclick=()=>{ idx=(idx+1)%ORDER_LINES.length; load(); };
  document.getElementById("woSpeak").onclick=()=>speak(ORDER_LINES[idx].zh);
  load();
})();

(function(){
  let order=[],pos=0,score=0,answered=false;
  document.getElementById("qTotal").textContent=QUIZ.length;
  function shuffle(a){a=a.slice();for(let i=a.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[a[i],a[j]]=[a[j],a[i]];}return a;}
  const els={ru:qRu,tag:qTag,hint:qHint,ans:qAns,zh:qZh,py:qPy,en:qEn,msg:qMsg,bar:qBar,play:qPlay,show:qShow};
  function start(){ order=shuffle(QUIZ.map((_,i)=>i)); pos=0; score=0; document.getElementById("qScore").textContent=0; next(); }
  function next(){
    answered=false;
    const q=QUIZ[order[pos]];
    els.ru.textContent=q.ru; els.hint.textContent="💡 "+q.hint;
    els.tag.textContent=`Вопрос ${pos+1} из ${QUIZ.length}`;
    els.ans.classList.remove("show"); els.play.style.display="none"; els.show.style.display="";
    els.msg.textContent=""; els.bar.style.width=(pos/QUIZ.length*100)+"%";
    els.zh.textContent=q.zh; els.py.textContent=q.py; els.en.textContent="";
    current=q;
  }
  let current=null;
  els.show.onclick=()=>{ els.ans.classList.add("show"); els.play.style.display=""; els.show.style.display="none"; speak(current.zh); };
  els.play.onclick=()=>speak(current.zh);
  function grade(ok){
    if(answered)return; answered=true;
    if(ok){score++;document.getElementById("qScore").textContent=score;
      els.msg.textContent="Красавчик! Идём дальше →";els.msg.className="msg ok";}
    else{els.msg.textContent="Запомни: "+current.zh+" — повтори вслух 2 раза";els.msg.className="msg bad";
      speak(current.zh,0.7);}
    pos++;
    if(pos>=QUIZ.length){
      els.bar.style.width="100%";
      setTimeout(()=>{
        els.ru.textContent=`Итог: ${score} из ${QUIZ.length}`; 
        els.ru.textContent= score>=QUIZ.length*0.8 ? "🏆 "+score+" / "+QUIZ.length+" — блестяще!" :
          score>=QUIZ.length*0.5 ? "💪 "+score+" / "+QUIZ.length+" — хорошо, ещё круг!" :
          "📖 "+score+" / "+QUIZ.length+" — повтори диалоги и попробуй снова";
        els.hint.textContent=""; els.ans.classList.remove("show"); els.show.style.display="none";
        els.tag.textContent="Финиш";
        els.show.style.display=""; els.show.textContent="🔁 Пройти ещё раз";
        els.show.onclick=start;
      },900);
      return;
    }
    setTimeout(next,1200);
  }
  document.getElementById("qOk").onclick=()=>grade(true);
  document.getElementById("qNo").onclick=()=>grade(false);
  start();
})();
